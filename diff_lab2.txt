+// Return the number of free pages.
+uint64 freepages(){
+  struct run *r;
+  uint64 n = 0;
+
+  acquire(&kmem.lock);
+
+  r = kmem.freelist;
+
+  while(r){
+    n += 1;
+    r = r->next;
+  }
+
+  release(&kmem.lock);
+
+  return n;
+}

+// Count processes whose state is not UNUSED.
+uint64
+nproc(void)
+{
+  struct proc *p;
+  uint64 n = 0;
+
+  for (p = proc; p < &proc[NPROC]; p++) {
+    acquire(&p->lock);
+    if(p->state != UNUSED){
+      n++;
+    }
+    release(&p->lock);
+  }
+  return n;
+}

+
+  printf("freepages = %lu pages\n", info.freepages);
+  printf("nproc     = %lu\n", info.nproc);
+
+  int pid = fork();
+
+  if (pid < 0) {
+    printf("fork failed\n");
+    exit(1);
+  }
+
+  if (pid == 0) {
+    // child
+    pause(20);
+    exit(0);
+  }
+
+  // parent
+  if (sysinfo(&info) < 0) {
+    printf("sysinfo failed\n");
+    exit(1);
+  }
+
+  printf("after fork, nproc = %lu\n", info.nproc);
+
+  wait(0);
+
+  exit(0);
